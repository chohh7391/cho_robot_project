cmake_minimum_required(VERSION 3.8)
project(cho_hardware_openarm_mit_mujoco)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(cho_openarm_mit_core REQUIRED)
find_package(mujoco_ros2_control REQUIRED)
find_package(pluginlib REQUIRED)
add_library(${PROJECT_NAME} SHARED src/mit_limiter.cpp src/mit_mujoco_system.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
ament_target_dependencies(${PROJECT_NAME}
  cho_openarm_mit_core mujoco_ros2_control pluginlib)
pluginlib_export_plugin_description_file(hardware_interface hardware.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(FILES hardware.xml DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 find_package(cho_description_openarm REQUIRED)
 find_program(XACRO_EXECUTABLE NAMES xacro REQUIRED)
 ament_add_gtest(test_mit_limiter test/test_mit_limiter.cpp)
 target_link_libraries(test_mit_limiter ${PROJECT_NAME})
 target_include_directories(test_mit_limiter PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 target_compile_definitions(test_mit_limiter PRIVATE OPENARM_SAFETY_PROFILE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../../cho_description/cho_description_openarm/config/mit_safety_profiles_v1.yaml")
 ament_add_gtest(test_plugin_load test/test_plugin_load.cpp)
 target_link_libraries(test_plugin_load ${PROJECT_NAME})
 ament_target_dependencies(test_plugin_load hardware_interface pluginlib)
 execute_process(COMMAND ${XACRO_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../cho_description/cho_description_openarm/robots/openarm_v10/openarm_v10.urdf.xacro hardware:=mujoco control_mode:=torque mujoco_mit_prototype:=true mujoco_mit_headless:=true OUTPUT_FILE ${CMAKE_CURRENT_BINARY_DIR}/mit_single.urdf RESULT_VARIABLE MIT_SINGLE_XACRO_RESULT)
 execute_process(COMMAND ${XACRO_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../cho_description/cho_description_openarm/robots/openarm_v10/openarm_v10.urdf.xacro hardware:=mujoco control_mode:=torque mujoco_mit_prototype:=true mujoco_mit_headless:=true bimanual:=true OUTPUT_FILE ${CMAKE_CURRENT_BINARY_DIR}/mit_bimanual.urdf RESULT_VARIABLE MIT_BIMANUAL_XACRO_RESULT)
 if(NOT MIT_SINGLE_XACRO_RESULT EQUAL 0)
   message(FATAL_ERROR "Failed to generate single-arm MIT wrapper fixture")
 endif()
 if(NOT MIT_BIMANUAL_XACRO_RESULT EQUAL 0)
   message(FATAL_ERROR "Failed to generate bimanual MIT wrapper fixture")
 endif()
 ament_add_gtest(test_resource_manager_headless test/test_resource_manager_headless.cpp)
 target_link_libraries(test_resource_manager_headless ${PROJECT_NAME})
 ament_target_dependencies(test_resource_manager_headless hardware_interface pluginlib)
 target_compile_definitions(test_resource_manager_headless PRIVATE MIT_SINGLE_URDF="${CMAKE_CURRENT_BINARY_DIR}/mit_single.urdf" MIT_BIMANUAL_URDF="${CMAKE_CURRENT_BINARY_DIR}/mit_bimanual.urdf" OPENARM_SAFETY_PROFILE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../../cho_description/cho_description_openarm/config/mit_safety_profiles_v1.yaml")
endif()
ament_package()
